set(VECTORUNE_HELPERS)
if(VECTORUNE_WITH_TILEMAKER)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tilemaker/CMakeLists.txt")
        message(FATAL_ERROR "Initialize tilemaker with: git submodule update --init tools/tilemaker (or set VECTORUNE_WITH_TILEMAKER=OFF)")
    endif()
    # Keep upstream compiler settings scoped to the converter, not the Qt app.
    function(add_tilemaker_engine)
        set(CMAKE_AUTOMOC OFF)
        add_subdirectory(tilemaker EXCLUDE_FROM_ALL)
    endfunction()
    add_tilemaker_engine()
    set_property(GLOBAL APPEND PROPERTY JOB_POOLS tilemaker_compile=2)
    set_property(TARGET tilemaker PROPERTY JOB_POOL_COMPILE tilemaker_compile)
    list(APPEND VECTORUNE_HELPERS tilemaker)
endif()
if(VECTORUNE_WITH_GARMIN)
    find_package(Qt6 REQUIRED COMPONENTS Network)
    add_executable(mapviewer_geometry_service garmin/geometry_main.cpp)
    target_link_libraries(mapviewer_geometry_service PRIVATE vectorune_geometry Qt6::Network)
    add_executable(mapviewer_garmin_import garmin/import_main.cpp)
    target_link_libraries(mapviewer_garmin_import PRIVATE mapviewer_garmin_attributes vectorune_core vectorune_geometry)
    add_executable(mapviewer_garmin_route garmin/route_main.cpp)
    target_link_libraries(mapviewer_garmin_route PRIVATE mapviewer_routing)
    add_executable(mapviewer_garmin_inspect garmin/inspect_main.cpp)
    target_compile_definitions(mapviewer_garmin_inspect PRIVATE DEBUG)
    target_compile_options(mapviewer_garmin_inspect PRIVATE -fno-operator-names)
    target_include_directories(mapviewer_garmin_inspect PRIVATE "${PROJECT_BINARY_DIR}/plugins/garmin")
    target_link_libraries(mapviewer_garmin_inspect PRIVATE mapviewer_garmin_attributes)
    list(APPEND VECTORUNE_HELPERS mapviewer_garmin_import mapviewer_garmin_route mapviewer_geometry_service)
    install(TARGETS mapviewer_garmin_inspect RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
endif()
