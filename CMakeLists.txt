cmake_minimum_required(VERSION 3.21)
project(VectoRune VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
option(VECTORUNE_BUILD_VIEWER "Build the Qt/MapLibre application" ON)
option(VECTORUNE_WITH_GARMIN "Build Garmin format adapter and tools" ON)
option(VECTORUNE_WITH_MBTILES "Build MBTiles format adapter and tools" ON)
option(MAPVIEWER_BUILD_TESTS "Compatibility alias for BUILD_TESTING" OFF)
option(BUILD_TESTING "Build regression tests" ${MAPVIEWER_BUILD_TESTS})
if(BUILD_TESTING)
    enable_testing()
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Positioning Sql)
add_subdirectory(core)
add_subdirectory(plugins)
add_subdirectory(tools)
if(VECTORUNE_BUILD_VIEWER)
    add_subdirectory(mapviewer)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
