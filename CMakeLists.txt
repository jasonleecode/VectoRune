cmake_minimum_required(VERSION 3.21)
project(VectoRune VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
option(VECTORUNE_BUILD_VIEWER "Build the Qt/MapLibre application" ON)
option(VECTORUNE_WITH_GARMIN "Build Garmin format adapter and tools" ON)
option(VECTORUNE_WITH_MBTILES "Build MBTiles format adapter and tools" ON)
option(VECTORUNE_WITH_TILEMAKER "Build the bundled OSM conversion engine" ON)
set(VECTORUNE_QT_INPUTCONTEXT_PLUGIN "" CACHE FILEPATH "Optional input method plugin built with this Qt version")
function(vectorune_deploy_inputcontext target)
    if(VECTORUNE_QT_INPUTCONTEXT_PLUGIN)
        if(NOT EXISTS "${VECTORUNE_QT_INPUTCONTEXT_PLUGIN}")
            message(FATAL_ERROR "Input method plugin not found: ${VECTORUNE_QT_INPUTCONTEXT_PLUGIN}")
        endif()
        set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${VECTORUNE_QT_INPUTCONTEXT_PLUGIN}")
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:${target}>/platforminputcontexts"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${VECTORUNE_QT_INPUTCONTEXT_PLUGIN}"
                "$<TARGET_FILE_DIR:${target}>/platforminputcontexts/")
    endif()
endfunction()
option(MAPVIEWER_BUILD_TESTS "Compatibility alias for BUILD_TESTING" OFF)
option(BUILD_TESTING "Build regression tests" ${MAPVIEWER_BUILD_TESTS})
if(BUILD_TESTING)
    enable_testing()
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Positioning Sql)
add_subdirectory(core)
add_subdirectory(plugins)
add_subdirectory(tools)
if(VECTORUNE_BUILD_VIEWER)
    add_subdirectory(mapviewer)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
